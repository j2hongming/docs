    #!/bin/bash
# ===================================================================
# Proxmox Cloud-Init VM Creator
# Usage: ./create-vm.sh <VMID> <NAME>
# Example: ./create-vm.sh 102 k8s-lab1-worker-2
# ===================================================================

if [ $# -ne 2 ]; then
  echo "Usage: $0 <VMID> <NAME>"
  echo "Example: $0 102 k8s-lab1-worker-2"
  exit 1
fi

VMID="$1"
NAME="$2"

# Check for CI_PASSWORD environment variable
if [ -z "${CI_PASSWORD}" ]; then
  echo "ERROR: CI_PASSWORD environment variable is not set!"
  echo "Please set it before running:"
  echo "  export CI_PASSWORD='your-strong-password-here'"
  exit 1
fi

echo "=== Creating VM ${VMID} - ${NAME} ==="

pvesh create /nodes/pve/qemu \
  --vmid "${VMID}" \
  --name "${NAME}" \
  --ostype l26 \
  --scsihw virtio-scsi-single \
  --cpu host \
  --sockets 4 \
  --cores 1 \
  --memory 8192 \
  --balloon 8192 \
  --net0 virtio,bridge=vmbr0,firewall=0 \
  --agent enabled=1,fstrim_cloned_disks=1 \
  --boot order=virtio0 \
  --onboot 0

pvesh set /nodes/pve/qemu/${VMID}/config \
  --ciuser root \
  --cipassword "${CI_PASSWORD}" \
  --ciupgrade 0 \
  --ipconfig0 "ip=dhcp"

qm importdisk ${VMID} noble-server-cloudimg-amd64.img local-lvm
qm set ${VMID} --virtio0 local-lvm:vm-${VMID}-disk-0
qm set ${VMID} --boot c --bootdisk virtio0
qm set ${VMID} --ide0 local-lvm:cloudinit

pvesh set /nodes/pve/qemu/${VMID}/resize --disk virtio0 --size 30G

echo "VM ${VMID} (${NAME}) created successfully!"

    #!/bin/bash
# ===================================================================
# Proxmox Cloud-Init VM Creator with Hashed Password
# Usage: ./create-vm.sh <VMID> <NAME>
# Example: ./create-vm.sh 102 k8s-lab1-worker-2
# ===================================================================

if [ $# -ne 2 ]; then
  echo "Usage: $0 <VMID> <NAME>"
  echo "Example: $0 102 k8s-lab1-worker-2"
  exit 1
fi

VMID="$1"
NAME="$2"

# Check for CI_PASSWORD
if [ -z "${CI_PASSWORD}" ]; then
  echo "ERROR: CI_PASSWORD environment variable is not set!"
  echo "Please set it before running:"
  echo "  export CI_PASSWORD='your-strong-password-here'"
  exit 1
fi

# Generate SHA-512 hashed password for cloud-init
echo "Generating hashed password..."
HASHED_PASSWD=$(openssl passwd -6 "${CI_PASSWORD}")

if [ -z "${HASHED_PASSWD}" ]; then
  echo "ERROR: Failed to generate password hash. Is 'openssl' installed?"
  echo "Install it with: apt install whois"
  exit 1
fi

echo "=== Creating VM ${VMID} - ${NAME} ==="

pvesh create /nodes/pve/qemu \
  --vmid "${VMID}" \
  --name "${NAME}" \
  --ostype l26 \
  --scsihw virtio-scsi-single \
  --cpu host \
  --sockets 4 \
  --cores 1 \
  --memory 8192 \
  --balloon 8192 \
  --net0 virtio,bridge=vmbr0,firewall=0 \
  --agent enabled=1,fstrim_cloned_disks=1 \
  --boot order=virtio0 \
  --onboot 0

# Cloud-Init configuration with hashed password
cat <<EOF > /var/lib/vz/snippets/cloud-init-config-${VMID}.yaml
#cloud-config
hostname: ${NAME}
ssh_pwauth: true
users:
  - name: root
    lock_passwd: false
    hashed_passwd: ${HASHED_PASSWD}
write_files:
  - path: /etc/ssh/sshd_config.d/99-allow-root-ssh.conf
    content: |
      PasswordAuthentication yes
      PermitRootLogin yes
    permissions: '0644'
EOF

pvesh set /nodes/pve/qemu/${VMID}/config \
  --ciupgrade 0 \
  --ipconfig0 "ip=dhcp" \
  --cicustom "user=local:snippets/cloud-init-config-${VMID}.yaml"

qm importdisk ${VMID} noble-server-cloudimg-amd64.img local-lvm
qm set ${VMID} --virtio0 local-lvm:vm-${VMID}-disk-0
qm set ${VMID} --boot c --bootdisk virtio0
qm set ${VMID} --ide0 local-lvm:cloudinit

pvesh set /nodes/pve/qemu/${VMID}/resize --disk virtio0 --size 30G

echo "VM ${VMID} (${NAME}) created successfully!"
echo "Cloud-init snippet: /var/lib/vz/snippets/cloud-init-config-${VMID}.yaml"

    enable the snippets 

    pvesm set local --content backup,images,iso,rootdir,snippets,vztmpl
